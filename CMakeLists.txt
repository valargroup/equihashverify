cmake_minimum_required(VERSION 3.18)
project(equihashverify LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 11)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
option(EQUIHASH_REQUIRE_AVX2 "Fail tests unless the AVX2 kernel is available and exercised" OFF)
option(EQUIHASH_REQUIRE_AVX512 "Require the experimental AVX-512 kernel in tests" OFF)
find_package(Threads REQUIRED)
find_path(SODIUM_INCLUDE_DIR sodium.h REQUIRED)
find_library(SODIUM_LIBRARY sodium REQUIRED)
find_path(BOOST_INCLUDE_DIR boost/static_assert.hpp REQUIRED)
add_library(equihashverify crypto/equihash.cpp crypto/equihash_200_9.cpp
            crypto/equihash_avx2.cpp crypto/equihash_avx512.cpp util.cpp)
target_include_directories(equihashverify PUBLIC . ${SODIUM_INCLUDE_DIR} ${BOOST_INCLUDE_DIR})
target_link_libraries(equihashverify PUBLIC ${SODIUM_LIBRARY} Threads::Threads)
# Keep the legacy signature self-test's assertion calls intact; the new verifier
# and its tests have no assertion-dependent work and build normally with NDEBUG.
set_source_files_properties(crypto/equihash.cpp PROPERTIES COMPILE_OPTIONS "-UNDEBUG")
include(CTest)
if(BUILD_TESTING)
  set(kernel_requirement "")
  if(EQUIHASH_REQUIRE_AVX2)
    set(kernel_requirement --require-avx2)
  endif()
  add_executable(verify_tests tests/verify.cpp)
  target_link_libraries(verify_tests PRIVATE equihashverify)
  add_test(NAME native_proofs COMMAND verify_tests ${CMAKE_CURRENT_SOURCE_DIR}/tests/proofs.tsv ${kernel_requirement})
  add_executable(avx2_hash_tests tests/avx2_hashes.cpp)
  target_link_libraries(avx2_hash_tests PRIVATE equihashverify)
  add_test(NAME leaf_hashes COMMAND avx2_hash_tests ${kernel_requirement})
  add_executable(portable_tests tests/portable.cpp)
  target_link_libraries(portable_tests PRIVATE equihashverify)
  add_test(NAME portable_vectors COMMAND portable_tests ${CMAKE_CURRENT_SOURCE_DIR}/tests/proofs.tsv ${CMAKE_CURRENT_SOURCE_DIR}/tests/hashes.tsv)
  set(avx512_requirement "")
  if(EQUIHASH_REQUIRE_AVX512)
    set(avx512_requirement --require-avx512)
  endif()
  add_executable(avx512_tests tests/avx512.cpp)
  target_link_libraries(avx512_tests PRIVATE equihashverify)
  add_test(NAME avx512_vectors COMMAND avx512_tests ${CMAKE_CURRENT_SOURCE_DIR}/tests/proofs.tsv ${avx512_requirement})
  add_executable(dispatch_tests tests/dispatch.cpp crypto/equihash_200_9.cpp)
  target_include_directories(dispatch_tests PRIVATE .)
  target_link_libraries(dispatch_tests PRIVATE Threads::Threads)
  foreach(backend scalar avx2 avx512)
    add_test(NAME dispatch_default_${backend} COMMAND ${CMAKE_COMMAND} -E env
      --unset=EQUIHASH_ENABLE_AVX512 $<TARGET_FILE:dispatch_tests>
      ${CMAKE_CURRENT_SOURCE_DIR}/tests/proofs.tsv ${backend} default)
    add_test(NAME dispatch_enabled_${backend} COMMAND ${CMAKE_COMMAND} -E env
      EQUIHASH_ENABLE_AVX512=1 $<TARGET_FILE:dispatch_tests>
      ${CMAKE_CURRENT_SOURCE_DIR}/tests/proofs.tsv ${backend} enabled)
  endforeach()
  add_test(NAME dispatch_unrecognized_setting COMMAND ${CMAKE_COMMAND} -E env
    EQUIHASH_ENABLE_AVX512=true $<TARGET_FILE:dispatch_tests>
    ${CMAKE_CURRENT_SOURCE_DIR}/tests/proofs.tsv avx512 default)
  find_package(Python3 COMPONENTS Interpreter REQUIRED)
  add_test(NAME reference_vectors COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tests/reference.py --check)
  set(EQUIHASH_RUST_REFERENCE_LIBRARY "" CACHE FILEPATH "Test-only pinned Rust reference built by tests/check_rust.py")
  if(EQUIHASH_RUST_REFERENCE_LIBRARY)
    add_executable(rust_differential tests/rust_differential.cpp)
    target_link_libraries(rust_differential PRIVATE equihashverify ${EQUIHASH_RUST_REFERENCE_LIBRARY})
    add_test(NAME rust_differential COMMAND rust_differential
      ${CMAKE_CURRENT_SOURCE_DIR}/tests/proofs.tsv ${kernel_requirement} ${avx512_requirement})
  endif()
endif()

option(EQUIHASH_BUILD_BENCHMARK "Build the standalone same-host benchmark" OFF)
if(EQUIHASH_BUILD_BENCHMARK)
  add_executable(pool_bench tests/benchmark.cpp)
  target_link_libraries(pool_bench PRIVATE equihashverify)
endif()
